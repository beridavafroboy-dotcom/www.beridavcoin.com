pragma solidity ^0.8.0;

contract BERIDAVToken {
    string public name     = "BERIDAV COIN";
    string public symbol   = "BER";
    uint   public totalSupply = 39_000_000_000;

    struct Innovation {
        string  title;
        address creator;
        uint    value;
    }
    mapping(uint => Innovation) public innovations;

    event NFTMinted(uint indexed id, string title, address creator);

    function mintInnovation(uint id, string calldata title) external {
        innovations[id] = Innovation(title, msg.sender, 0);
        emit NFTMinted(id, title, msg.sender);
    }
